using Oracle.ManagedDataAccess.Client;
using System.Collections.Generic;
using System.Data;
using System.Data.Common;
namespace EjemCodigo
{
    public class EjemploTabla
    {
        public int id { get; set; }
        public string nombre { get; set; }


        public List<EjemploTabla> Get()
        {
            var lista = new List<EjemploTabla>();

            var conexion = new OracleConnection("Data Source=(DESCRIPTION=(ADDRESS=(PROTOCOL=TCP)(HOST=10.1.4.15)(PORT=1521))(CONNECT_DATA=(SERVICE_NAME=VACDES))); User Id=VAC;Password=proyecto2020");
            conexion.Open();

            var cmd = new OracleCommand();
            cmd.Connection = conexion;
            cmd.CommandText = "SELECT id, nomrbe From EJEMPLO";

            using (IDataReader dr = cmd.ExecuteReader())
            {
                while (dr.Read())
                {
                    var item = new EjemploTabla();
                    item.id = int.Parse(dr.GetValue(0).ToString());
                    item.nombre = dr.GetValue(1).ToString();
                    lista.Add(item);
                }
            }
            conexion.Close();
            conexion.Dispose();

            return lista;



        }
    }
}
